const esc = s => String(s ?? '').replace(/[&<>"']/g, c => ({
  '&': '&amp;',
  '<': '&lt;',
  '>': '&gt;',
  '"': '&quot;',
  "'": '&#39;'
})[c]);
const money = n => new Intl.NumberFormat('en-PK', {
  maximumFractionDigits: 0
}).format(n);
const lakh = n => (n / 100000).toFixed(2);
const car = s => cars.find(c => c.slug === s);
const link = u => /^https:\/\//.test(u) ? esc(u) : '#';
const vFor = c => c.variants[variants[c.slug] || 0];
const keyFor = c => c.slug + ':' + (variants[c.slug] || 0);
const priceFor = c => overrides[keyFor(c)] ?? M.midpoint(vFor(c));
const label = c => `${c.brand} ${c.name} ${c.condition === 'used' ? c.year : '· New'}`;
const range = (lo, hi) => lakh(lo) + (hi > lo ? '–' + lakh(hi) : '') + ' lakh';
let toastTimer;

}

}
function photoErrors() {

  });
}

function renderBrands() {
  const names = ['', 'Toyota', 'Honda', 'Suzuki', 'Hyundai'];
  $('#brandTabs').innerHTML = names.map(n => `<button role="tab" aria-selected="${brand === n}" class="brand-tab" data-brand="${n}">${n || 'All companies'}<span>${cars.filter(c => !n || c.brand === n).length} cars</span></button>`).join('');
}
function render() {
  const f = {
    q: $('#search').value.trim(),
    brand,
    body,
    condition: $('#condition').value,
    year: $('#year').value,
    budget: M.num($('#budget').value) * 100000,
    savedOnly,
    saved
  };
  let list = cars.filter(c => M.matches(c, f));
  const sort = $('#sort').value;
  if (sort === 'low') list.sort((a, b) => M.minPrice(a) - M.minPrice(b));
  if (sort === 'high') list.sort((a, b) => M.minPrice(b) - M.minPrice(a));
  if (sort === 'name') list.sort((a, b) => label(a).localeCompare(label(b)));
  $('#resultCount').textContent = `${list.length} entries`;
  $('#savedNotice').hidden = !savedOnly;
  $('#savedCount').textContent = saved.length;
  $('#compareCount').textContent = compare.length;
  renderBrands();

'on' : ''}" data-save="${c.slug}" aria-label="${saved.includes(c.slug) ? 'Unsave' : 'Save'} ${esc(label(c))}" aria-pressed="${saved.includes(c.slug)}">${saved.includes(c.slug) ? '♥' : '♡'}</button><img src="${esc(c.image)}" alt="${esc(label(c))}" loading="lazy"><div class="image-fallback"><strong>${esc(c.name)}</strong><small>${c.year} · Photo unavailable</small></div></div><div class="car-info"><span class="condition-pill ${c.condition}">${c.condition === 'used' ? 'Used · ' + c.year : 'New car'}</span><div class="car-brand">${esc(c.brand)}</div><div class="car-title"><h3>${esc(c.name)}<small>${c.condition === 'used' ? c.year + ' model' : c.year + ' catalogue'}</small></h3><span>${c.variants.length > 1 ? c.variants.length + ' variants' : ''}</span></div><div class="spec-mini"><span>◈ ${esc(c.variants[0].engine.split(' ·')[0])}</span><span>◇ ${esc(c.variants[0].fuel)}</span></div><div class="car-price"><small>${c.condition === 'used' ? 'OBSERVED ASKING PRICES' : 'STARTING FROM'}</small><strong>${c.condition === 'used' ? range(M.minPrice(c), M.maxPrice(c)) : '<span>PKR</span> ' + money(M.minPrice(c))}</strong><div class="range">${c.condition === 'used' ? function updateCosts() {

}

(function (root) {
  'use strict';

  const minPrice = c => Math.min(...c.variants.map(v => v.price));
  const maxPrice = c => Math.max(...c.variants.map(v => v.price_max ?? v.price));
  const num = (x, f = 0) => Number.isFinite(Number(x)) && Number(x) >= 0 ? Number(x) : f;
  const midpoint = v => (v.price + (v.price_max ?? v.price)) / 2;
  function upfront(c, price, fees, reserve) {
    return num(price) + num(fees) + (c.condition === 'used' ? num(reserve) : 0);
  }
  function budgetStatus(c, budget, fees, reserve) {
    const low = upfront(c, minPrice(c), fees, reserve),
      high = upfront(c, maxPrice(c), fees, reserve);
    return {
      low,
      high,
      fit: high <= budget ? 'within' : low <= budget ? 'partial' : 'over'
    };
  }
  function fuelCost(km, price, economy) {
    return Number(economy) > 0 && Number.isFinite(Number(economy)) ? num(km) * num(price) / Number(economy) : null;
  }
  function matches(c, f) {
    return (!f.q || (c.brand + ' ' + c.name + ' ' + c.year).toLowerCase().includes(f.q.toLowerCase())) && (!f.brand || c.brand === f.brand) && (!f.body || c.body === f.body) && (!f.condition || c.condition === f.condition) && (!f.year || c.year === Number(f.year)) && (!f.budget || minPrice(c) <= f.budget) && (!f.savedOnly || f.saved.includes(c.slug));
  }
  function shortlist(cars, p) {
    return cars.map(c => ({
      c,
      ...budgetStatus(c, p.budget, p.fees, p.reserve)
    })).filter(x => x.fit !== 'over').sort((a, b) => {
      const score = x => {
        let s = x.fit === 'within' ? 4 : 0;
        if (p.priority === 'new' && x.c.condition === 'new') s += 10;
        if (p.priority === 'family' && x.c.body === 'Sedan') s += 10;
        if (p.priority === 'city' && x.c.body === 'Hatchback') s += 10;
        return s;
      };
      return score(b) - score(a) || a.low - b.low;
    }).slice(0, 3);
  }
  const api = {
    minPrice,
    maxPrice,
    num,
    midpoint,
    upfront,
    budgetStatus,
    fuelCost,
    matches,
    shortlist
  };
  if (typeof module !== 'undefined') module.exports = api;else root.CarMath = api;
})(typeof window !== 'undefined' ? window : globalThis);

"""Run PakAutoHub locally using Python standard library only."""
from functools import partial
from http.server import SimpleHTTPRequestHandler, ThreadingHTTPServer
from pathlib import Path
import sys
import threading
import webbrowser

def main():
    try:
        port = int(sys.argv[1]) if len(sys.argv) > 1 else 8000
        if not 1 <= port <= 65535:
            raise ValueError
    except ValueError:
        print("Use a port from 1 to 65535: python run_local.py 8001")
        return 1
    folder = Path(__file__).resolve().parent
    handler = partial(SimpleHTTPRequestHandler, directory=str(folder))
    try:
        server = ThreadingHTTPServer(("127.0.0.1", port), handler)
    except OSError as exc:
        print(f"Could not start: {exc}. Try another port: python run_local.py 8001")
        return 1
    url = f"http://127.0.0.1:{port}"
    print(f"PakAutoHub: {url}\nKeep this terminal open. Press Ctrl+C to stop.")
    timer = threading.Timer(0.7, webbrowser.open, args=(url,))
    timer.daemon = True
    timer.start()
    try:
        server.serve_forever()
    except KeyboardInterrupt:
        print("\nStopped.")
    finally:
        timer.cancel()
        server.server_close()
    return 0

if __name__ == "__main__":
    raise SystemExit(main())
